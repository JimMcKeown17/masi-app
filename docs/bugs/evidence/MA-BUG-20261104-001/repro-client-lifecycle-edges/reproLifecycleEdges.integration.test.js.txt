const RELATIONS = {

const VERSIONED = new Set([

const createFakeServer = (initial = {}) => {
  const T = {};
  const table = (name) => { if (!T[name]) T[name] = new Map(); return T[name]; };
  for (const [name, rows] of Object.entries(initial)) {
    for (const row of rows) table(name).set(row.id, { ...(VERSIONED.has(name) ? { row_version: 1 } : {}), ...row });
  }
  const calls = [];
  const interceptors = [];
  const all = (name) => [...table(name).values()];
  const intercept = (fn) => interceptors.push(fn);
  const runInterceptors = (op) => {
    for (let i = 0; i < interceptors.length; i += 1) {
      const outcome = interceptors[i](op);
      if (outcome) {
        if (outcome.once !== false) interceptors.splice(i, 1);
        return outcome;
      }
    }
    return null;
  };

  const clientFor = (actor) => {
    // ---- RLS helpers (pre-statement snapshot), modelled on 20260521144901 + later migrations.
    const writeForChild = (childId) => (
      all('child_ea_assignments').some(a => a.child_id === childId && a.user_id === actor && a.unassigned_at == null)
      || all('class_ea_assignments').some(ca => ca.ea_user_id === actor && ca.unassigned_at == null
        && all('child_class_memberships').some(m => m.child_id === childId && m.class_id === ca.class_id && m.exited_at == null))
      || all('group_ea_assignments').some(ga => ga.ea_user_id === actor && ga.unassigned_at == null
        && all('child_group_memberships').some(m => m.child_id === childId && m.group_id === ga.group_id && m.removed_at == null))
    );
    const writeForClass = (classId) => table('classes').get(classId)?.created_by === actor
      || all('class_ea_assignments').some(ca => ca.class_id === classId && ca.ea_user_id === actor && ca.unassigned_at == null);
    const accessClass = (classId) => {
      const c = table('classes').get(classId);
      if (!c) return false;
      return c.created_by === actor
        || all('class_ea_assignments').some(ca => ca.class_id === classId && ca.ea_user_id === actor)
        || all('staff_programme_assignments').some(s => s.user_id === actor && s.ended_at == null && s.school_id === c.school_id);
    };
    const writeForGroup = (groupId) => table('groups').get(groupId)?.created_by === actor
      || all('group_ea_assignments').some(g => g.group_id === groupId && g.ea_user_id === actor && g.unassigned_at == null);
    const accessGroup = (groupId) => table('groups').get(groupId)?.created_by === actor
      || all('group_ea_assignments').some(g => g.group_id === groupId && g.ea_user_id === actor);
    const readChild = (childId) => {
      const c = table('children').get(childId);
      if (!c) return false;
      return c.created_by === actor
        || all('child_ea_assignments').some(a => a.child_id === childId && a.user_id === actor)
        || writeForChild(childId);
    };
    const openProgramme = (programmeId) => all('staff_programme_assignments')
      .some(s => s.user_id === actor && s.programme_id === programmeId && s.ended_at == null);

    const canSelect = {
      staff_programme_assignments: r => r.user_id === actor,
      academic_years: () => true,
      children: r => readChild(r.id),
      child_ea_assignments: r => r.user_id === actor,
      child_programme_enrollments: r => readChild(r.child_id),
      child_class_memberships: r => readChild(r.child_id) || accessClass(r.class_id),
      classes: r => accessClass(r.id),
      class_ea_assignments: r => r.ea_user_id === actor,
      groups: r => accessGroup(r.id),
      group_ea_assignments: r => r.ea_user_id === actor,
      child_group_memberships: r => readChild(r.child_id) || accessGroup(r.group_id),
    };
    const canUpdate = {
      classes: r => writeForClass(r.id),
      children: r => r.created_by === actor || writeForChild(r.id),
      child_ea_assignments: r => r.user_id === actor,
      class_ea_assignments: r => r.ea_user_id === actor,
      group_ea_assignments: r => r.ea_user_id === actor,
      child_programme_enrollments: r => writeForChild(r.child_id),
      child_class_memberships: r => writeForChild(r.child_id) && accessClass(r.class_id),
      child_group_memberships: r => writeForChild(r.child_id) && writeForGroup(r.group_id),
      groups: r => writeForGroup(r.id),
    };
    const canInsert = {
      children: r => r.created_by === actor,
      child_ea_assignments: r => r.user_id === actor && r.created_by === actor
        && table('children').get(r.child_id)?.created_by === actor,
      child_programme_enrollments: r => r.created_by === actor && writeForChild(r.child_id),
      child_class_memberships: r => r.created_by === actor && writeForChild(r.child_id) && accessClass(r.class_id),
      child_group_memberships: r => r.created_by === actor && writeForChild(r.child_id) && writeForGroup(r.group_id),
      classes: r => r.created_by === actor,
      class_ea_assignments: r => r.ea_user_id === actor && accessClass(r.class_id) && openProgramme(r.programme_id),
      groups: r => r.created_by === actor && openProgramme(r.programme_id),
      group_ea_assignments: r => r.ea_user_id === actor && table('groups').get(r.group_id)?.created_by === actor
        && openProgramme(r.programme_id),
      sessions: r => r.user_id === actor && openProgramme(r.programme_id),
      session_attendees: r => writeForChild(r.child_id),
    };
    const uniqueViolation = (name, r) => {
      if (name === 'child_group_memberships' && r.removed_at == null && r.grouping_version_id != null) {
        return all(name).some(o => o.id !== r.id && o.removed_at == null && o.child_id === r.child_id
          && o.grouping_version_id === r.grouping_version_id) && 'idx_child_group_memberships_active_by_version';
      }
      if (name === 'child_class_memberships' && r.exited_at == null) {
        return all(name).some(o => o.id !== r.id && o.exited_at == null && o.child_id === r.child_id
          && o.academic_year_id === r.academic_year_id) && 'idx_child_class_memberships_active_unique';
      }
      if (name === 'group_ea_assignments' && r.unassigned_at == null) {
        return all(name).some(o => o.id !== r.id && o.unassigned_at == null && o.group_id === r.group_id)
          && 'idx_group_ea_assignments_active_unique';
      }
      return false;
    };
    const rls = (name) => ({ code: '42501', message: `new row violates row-level security policy for table "${name}"` });

    const matches = (row, filters) => filters.every(({ kind, c, v }) => {
      if (kind === 'eq') return row[c] === v;
      if (kind === 'is') return v === null ? row[c] == null : row[c] === v;
      if (kind === 'in') return v.includes(row[c]);
      return true;
    });

    const executeSelect = (state) => {
      const baseFilters = state.filters.filter(f => !f.c.includes('.'));
      const embeds = [...String(state.columns).matchAll(/(\w+)(!inner)?\(([^)]*)\)/g)]
        .map(([, rel, inner]) => ({ rel, inner: Boolean(inner) }));
      const plain = String(state.columns).replace(/(\w+)(!inner)?\(([^)]*)\)/g, '')
        .split(',').map(s => s.trim()).filter(Boolean);
      let rows = all(state.name).filter(r => (canSelect[state.name] || (() => true))(r) && matches(r, baseFilters));
      const out = [];
      for (const base of rows) {
        let keep = true;
        const next = { ...base };
        for (const { rel, inner } of embeds) {
          const spec = RELATIONS[`${state.name}.${rel}`];
          if (!spec) throw new Error(`fake server: unknown relation ${state.name}.${rel}`);
          const relFilters = state.filters.filter(f => f.c.startsWith(`${rel}.`))
            .map(f => ({ ...f, c: f.c.slice(rel.length + 1) }));
          const related = all(spec.table).filter(r => r[spec.foreign] === base[spec.local]
            && (canSelect[spec.table] || (() => true))(r) && matches(r, relFilters));
          if (inner && related.length === 0) keep = false;
          next[rel] = spec.one ? (related[0] ? { ...related[0] } : null) : related.map(r => ({ ...r }));
        }
        if (!keep) continue;
        if (!plain.includes('*') && plain.length > 0) {
          const projected = {};
          for (const c of plain) projected[c] = next[c];
          for (const { rel } of embeds) projected[rel] = next[rel];
          out.push(projected);
        } else out.push(next);
      }
      rows = out.slice(0, state.limitN);
      return { data: rows, error: null };
    };

    const executeUpdate = (state) => {
      const op = { type: 'update', table: state.name, filters: state.filters, patch: state.patch, actor };
      calls.push(op);
      const forced = runInterceptors(op);
      if (forced?.throw) throw forced.throw;
      if (forced?.result) { op.result = forced.result; return forced.result; }
      const target = all(state.name).filter(r => matches(r, state.filters));
      const visible = target.filter(r => (canSelect[state.name] || (() => true))(r)
        && (canUpdate[state.name] || (() => true))(r));
      for (const r of visible) {
        if (VERSIONED.has(state.name) && typeof state.patch.row_version === 'number'
          && state.patch.row_version < r.row_version) {
          op.result = { error: { code: 'MV001', message: 'stale_version' } };
          return { data: null, error: { code: 'MV001', message: 'stale_version' } };
        }
      }
      for (const r of visible) table(state.name).set(r.id, { ...r, ...state.patch });
      const result = { data: state.wantSelect ? visible.map(r => ({ id: r.id })) : null, error: null };
      op.result = { rowsAffected: visible.length };
      return result;
    };

    const doUpsert = async (name, payload, options = {}) => {
      const rows = Array.isArray(payload) ? payload : [payload];
      const op = { type: 'upsert', table: name, ids: rows.map(r => r.id), ignoreDuplicates: Boolean(options.ignoreDuplicates), actor };
      calls.push(op);
      const forced = runInterceptors({ ...op, rows });
      if (forced?.throw) throw forced.throw;
      if (forced?.result) { op.result = forced.result; return forced.result; }
      // Validate the whole statement first (PostgREST bulk upsert is one atomic statement).
      const staged = new Map();
      for (const r of rows) {
        const existing = table(name).get(r.id) || staged.get(r.id);
        const insertOk = (canInsert[name] || (() => true))(r);
        if (!insertOk) { op.result = { error: rls(name) }; return { error: rls(name) }; }
        if (existing && options.ignoreDuplicates) continue;
        if (existing && !(canUpdate[name] || (() => true))(existing)) {
          const e = { code: '42501', message: `new row violates row-level security policy (USING expression) for table "${name}"` };
          op.result = { error: e };
          return { error: e };
        }
        const merged = { ...(VERSIONED.has(name) ? { row_version: 1 } : {}), ...existing, ...r };
        // Unique check sees committed + earlier rows in this statement.
        const saved = new Map(table(name));
        for (const [k, v] of staged) table(name).set(k, v);
        const violated = uniqueViolation(name, merged);
        T[name] = saved;
        if (violated) {
          const e = { code: '23505', message: `duplicate key value violates unique constraint "${violated}"` };
          op.result = { error: e };
          return { error: e };
        }
        staged.set(r.id, merged);
      }
      for (const [k, v] of staged) table(name).set(k, v);
      op.result = { applied: staged.size };
      return { data: null, error: null };
    };

    const hasHistory = (childId) => (
      ['session_attendees', 'assessments', 'letter_mastery', 'skill_mastery', 'battery_runs', 'child_group_memberships']
        .some(n => all(n).some(r => r.child_id === childId))
      || all('child_ea_assignments').some(r => r.child_id === childId && r.unassigned_at != null)
      || all('child_programme_enrollments').some(r => r.child_id === childId && r.ended_at != null)
      || all('child_class_memberships').some(r => r.child_id === childId && r.exited_at != null)
    );

    const acknowledgments = () => {
      const spa = all('staff_programme_assignments').filter(s => s.user_id === actor && s.ended_at == null)[0];
      const prog = spa?.programme_id || null;
      const ca = all('child_ea_assignments').filter(a => a.user_id === actor && a.unassigned_at == null);
      const visible = [...new Set(ca.map(a => a.child_id).filter(childId => all('child_programme_enrollments')
        .some(e => e.child_id === childId && e.programme_id === prog && e.ended_at == null)))];
      const enrollments = all('child_programme_enrollments').filter(e => ca.some(a => a.child_id === e.child_id)
        && e.programme_id === prog && e.ended_at == null);
      const memberships = all('child_class_memberships').filter(m => visible.includes(m.child_id) && m.exited_at == null);
      const classAssignments = all('class_ea_assignments').filter(a => a.ea_user_id === actor && a.programme_id === prog && a.unassigned_at == null);
      const roster = all('child_class_memberships').filter(m => classAssignments.some(a => a.class_id === m.class_id) && m.exited_at == null);
      const groupAssignments = all('group_ea_assignments').filter(g => g.ea_user_id === actor && g.programme_id === prog
        && g.unassigned_at == null && table('groups').get(g.group_id)?.programme_id === prog);
      const groupMemberships = all('child_group_memberships').filter(m => groupAssignments.some(g => g.group_id === m.group_id) && m.removed_at == null);
      const ids = rows => rows.map(r => r.id).sort();
      return {
        schema_version: 1, complete: true, user_id: actor, generated_at: new Date().toISOString(),
        active_programme_id: prog,
        child_ea_assignment_ids: ids(ca), assigned_child_ids: ca.map(a => a.child_id).sort(),
        visible_child_ids: visible.sort(), child_programme_enrollment_ids: ids(enrollments),
        child_class_membership_ids: ids(memberships), class_ea_assignment_ids: ids(classAssignments),
        class_ids: classAssignments.map(a => a.class_id).sort(), roster_class_ids: classAssignments.map(a => a.class_id).sort(),
        roster_membership_ids: ids(roster), group_ea_assignment_ids: ids(groupAssignments),
        group_ids: groupAssignments.map(g => g.group_id).sort(), child_group_membership_ids: ids(groupMemberships),
      };
    };

    const rpc = async (fn, args) => {
      const op = { type: 'rpc', fn, args, actor };
      calls.push(op);
      const forced = runInterceptors(op);
      if (forced?.throw) throw forced.throw;
      if (forced?.result) { op.result = forced.result; return forced.result; }
      if (fn === 'get_reconcile_acknowledgments') return { data: acknowledgments(), error: null };
      if (fn === 'delete_child_if_no_history') {
        const id = args.p_child_id;
        const child = table('children').get(id);
        if (!child) { op.result = { data: true }; return { data: true, error: null }; }
        if (!(child.created_by === actor || writeForChild(id))) {
          const e = { code: '42501', message: `Not authorized to delete child ${id}` };
          op.result = { error: e };
          return { data: null, error: e };
        }
        if (hasHistory(id)) { op.result = { data: false }; return { data: false, error: null }; }
        for (const n of ['child_class_memberships', 'child_programme_enrollments', 'child_ea_assignments']) {
          for (const r of all(n)) if (r.child_id === id) table(n).delete(r.id);
        }
        table('children').delete(id);
        op.result = { data: true };
        return { data: true, error: null };
      }
      return { data: null, error: { code: 'PGRST202', message: `unknown rpc ${fn}` } };
    };

    const from = (name) => {
      const state = { name, op: 'select', columns: '*', filters: [], limitN: Infinity, patch: null, wantSelect: false };
      const b = {
        select(cols = '*') { if (state.op === 'update') state.wantSelect = true; else state.columns = cols; return b; },
        update(patch) { state.op = 'update'; state.patch = patch; return b; },
        eq(c, v) { state.filters.push({ kind: 'eq', c, v }); return b; },
        is(c, v) { state.filters.push({ kind: 'is', c, v }); return b; },
        in(c, v) { state.filters.push({ kind: 'in', c, v }); return b; },
        order() { return b; },
        limit(n) { state.limitN = n; return b; },
        upsert: (payload, options) => doUpsert(name, payload, options),
        then(resolve, reject) {
          return Promise.resolve()
            .then(() => (state.op === 'update' ? executeUpdate(state) : executeSelect(state)))
            .then(resolve, reject);
        },
      };
      return b;
    };
    return { from, rpc, auth: {} };
  };

  return { table, all, calls, intercept, clientFor };
};

const insertLocal = async (db, tableName, row) => {
  const record = { ...row, sync_status: 'synced' };
  const keys = Object.keys(record);
  await db.runAsync(
    `insert into ${tableName} (${keys.join(', ')}) values (${keys.map(() => '?').join(', ')})`,
    ...keys.map(k => record[k]),
  );
};

      : `NOT_OK:${s.failureKind}`])),
  };
};

const callSummary = (server, from = 0) => server.calls.slice(from)
  .filter(c => c.type !== 'select')
  .map(c => {
    if (c.type === 'rpc') return `rpc:${c.fn}${c.args?.p_child_id ? `(${c.args.p_child_id})` : ''} -> ${JSON.stringify(c.result || '')}`;
    if (c.type === 'update') {
      const id = c.filters.find(f => f.c === 'id')?.v;
      return `update:${c.table}:${id} ${JSON.stringify(Object.keys(c.patch))} -> ${JSON.stringify(c.result || '')}`;
    }
    return `upsert:${c.table}:${c.ids.join('+')}${c.ignoreDuplicates ? '(ignoreDuplicates)' : ''} -> ${JSON.stringify(c.result || '')}`;
  });

describe('E5 shared class archived by a co-EA', () => {
  test('E5 EA_A archives shared class; EA_B phone pulls while EA_B class assignment is still active', async () => {
    const server = createFakeServer(baseServerRows());
    const phoneA = await createPhone(EA);
    const phoneB = await createPhone(EA_B);
    for (const phone of [phoneA, phoneB]) {
      await insertLocal(phone.db, 'classes', klass('class-shared'));
    }
    server.table('classes').set('class-shared', { row_version: 1, ...klass('class-shared') });
    await seedBoth(phoneA, server, 'class_ea_assignments', classAssignment('cea-a-s', 'class-shared', EA));
    await seedBoth(phoneB, server, 'class_ea_assignments', classAssignment('cea-b-s', 'class-shared', EA_B));
    // EA_A's delivery child c8, EA_B's delivery child c7, both in the shared class.
    for (const [phone, id, ea] of [[phoneA, 'child-8', EA], [phoneB, 'child-7', EA_B]]) {
      await seedBoth(phone, server, 'children', child(id, 'class-shared'));
      await seedBoth(phone, server, 'child_ea_assignments', childAssignment(`cha-${id}`, ea, id));
      await seedBoth(phone, server, 'child_programme_enrollments', enrollment(`cpe-${id}`, id));
      await seedBoth(phone, server, 'child_class_memberships', classMembership(`ccm-${id}`, id, 'class-shared'));
    }
    // EA_B also sees c8 as a class-only (control) child via its roster.
    for (const row of [child('child-8', 'class-shared')]) await insertLocal(phoneB.db, 'children', row);
    await insertLocal(phoneB.db, 'child_class_memberships', classMembership('ccm-child-8', 'child-8', 'class-shared'));
    record('E5', 'EA_B-before', { readers: await readers(phoneB) });

    await phoneA.repositories.classes.archiveClass('class-shared', { actorUserId: EA });
    const passA = await engineFor(phoneA, server).syncAll();
    record('E5', 'EA_A-pass', {
      result: { success: passA.success, totalTerminal: passA.totalTerminal },
      calls: callSummary(server), outbox: await outbox(phoneA),
      server: {
        class: server.table('classes').get('class-shared'),
        ceaA: server.table('class_ea_assignments').get('cea-a-s'),
        ceaB: server.table('class_ea_assignments').get('cea-b-s'),
        ccm7: server.table('child_class_memberships').get('ccm-child-7'),
        ccm8: server.table('child_class_memberships').get('ccm-child-8'),
        cha7: server.table('child_ea_assignments').get('cha-child-7'),
      },
    });
    const pulledB = await pullAll(phoneB, server);
    record('E5', 'EA_B-after-pull', {
      pulled: pulledB,
      localClass: await localRow(phoneB, 'classes', 'class-shared', 'archived_at, sync_status'),
      localCeaB: await localRow(phoneB, 'class_ea_assignments', 'cea-b-s', 'unassigned_at, sync_status'),
      localCha7: await localRow(phoneB, 'child_ea_assignments', 'cha-child-7', 'unassigned_at, sync_status'),
      localCcm7: await localRow(phoneB, 'child_class_memberships', 'ccm-child-7', 'exited_at, sync_status'),
      readers: await readers(phoneB),
      getClassById: (await phoneB.repositories.classes.getClassById('class-shared'))?.archived_at,
      getChildren_all: (await phoneB.repositories.children.getChildren()).map(c => c.id),
      outbox: await outbox(phoneB),
    });
    expect(true).toBe(true);
  });
});
