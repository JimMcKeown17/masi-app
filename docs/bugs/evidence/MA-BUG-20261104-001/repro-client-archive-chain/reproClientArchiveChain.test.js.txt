const createFakePostgrest = ({ actor }) => {
  const rows = {};
  const t = (name) => { if (!rows[name]) rows[name] = new Map(); return rows[name]; };
  const all = (name) => [...t(name).values()];
  const log = [];
  const receipts = new Map();

  const writeForChild = (childId) => (
    all('child_ea_assignments').some(r => r.child_id === childId && r.user_id === actor && r.unassigned_at == null)
    || all('class_ea_assignments').some(cls => cls.ea_user_id === actor && cls.unassigned_at == null
      && all('child_class_memberships').some(m => m.class_id === cls.class_id && m.child_id === childId && m.exited_at == null))
    || all('group_ea_assignments').some(g => g.ea_user_id === actor && g.unassigned_at == null
      && all('child_group_memberships').some(m => m.group_id === g.group_id && m.child_id === childId && m.removed_at == null))
  );
  const deliversToChild = (childId) => all('child_ea_assignments')
    .some(r => r.child_id === childId && r.user_id === actor && r.unassigned_at == null);
  const writeForGroup = (groupId) => t('groups').get(groupId)?.created_by === actor
    || all('group_ea_assignments').some(r => r.group_id === groupId && r.ea_user_id === actor && r.unassigned_at == null);
  const writeForClass = (classId) => t('classes').get(classId)?.created_by === actor
    || all('class_ea_assignments').some(r => r.class_id === classId && r.ea_user_id === actor && r.unassigned_at == null);
  const accessClass = (classId) => {
    const c = t('classes').get(classId);
    if (!c) return false;
    return c.created_by === actor
      || all('staff_programme_assignments').some(s => s.user_id === actor && s.school_id === c.school_id && s.ended_at == null)
      || all('class_ea_assignments').some(r => r.class_id === classId && r.ea_user_id === actor);
  };
  const heldGroup = (groupId, programmeId) => t('groups').get(groupId)?.programme_id === programmeId
    && all('group_ea_assignments').some(r => r.group_id === groupId && r.ea_user_id === actor);
  const openProgramme = (programmeId) => all('staff_programme_assignments')
    .some(s => s.user_id === actor && s.programme_id === programmeId && s.ended_at == null);

  const rls = (table, kind = '') => ({
    status: 403,
    body: { code: '42501', details: null, hint: null,
      message: `new row violates row-level security policy${kind} for table "${table}"` },
  });

  // INSERT WITH CHECK, per migration files (see report notes for file:line).
  const insertCheck = (table, row) => {
    if (table === 'sessions') {
      if (row.user_id !== actor || !openProgramme(row.programme_id)) return rls(table);
      const state = row.state ?? 'completed';
      if (state !== 'completed') return rls(table);
      if (row.group_id != null && !heldGroup(row.group_id, row.programme_id)) return rls(table);
      return null;
    }
    if (table === 'session_attendees') {
      const session = t('sessions').get(row.session_id);
      if (!session || session.user_id !== actor) return rls(table);
      if (!writeForChild(row.child_id)) return rls(table);
      if (row.group_id != null && !heldGroup(row.group_id, session.programme_id)) return rls(table);
      return null;
    }
    return { status: 500, body: { code: 'NOT_MODELLED', message: `insert into ${table} not modelled` } };
  };
  // UPDATE USING (old row) and WITH CHECK (new row); helpers read the pre-statement snapshot.
  const updatePolicy = (table, row) => {
    switch (table) {
      case 'children': return row.created_by === actor || writeForChild(row.id);
      case 'child_programme_enrollments': return writeForChild(row.child_id);
      case 'child_class_memberships': return writeForChild(row.child_id) && accessClass(row.class_id);
      case 'child_group_memberships': return writeForChild(row.child_id) && writeForGroup(row.group_id);
      case 'child_ea_assignments': return row.user_id === actor;
      case 'class_ea_assignments':
      case 'group_ea_assignments': return row.ea_user_id === actor;
      case 'groups': return writeForGroup(row.id);
      case 'classes': return writeForClass(row.id);
      default: return null;
    }
  };

  const flags = { attendeesDown: false };
  const upsert = (table, body, prefer) => {
    if (table === 'session_attendees' && flags.attendeesDown) {
      return { status: 503, body: { message: 'upstream request timeout' }, rowsAffected: 0 };
    }
    const proposed = Array.isArray(body) ? body : [body];
    const ignore = prefer.includes('resolution=ignore-duplicates');
    for (const row of proposed) {
      const error = insertCheck(table, row);
      if (error) return { ...error, rowsAffected: 0, firstFailingId: row.id };
    }
    for (const row of proposed) {
      if (ignore && t(table).has(row.id)) continue;
      t(table).set(row.id, { ...t(table).get(row.id), ...row });
    }
    return { status: 201, body: undefined, rowsAffected: proposed.length };
  };

  const patch = (table, body, params) => {
    const idFilter = params.get('id') || '';
    const id = idFilter.replace(/^eq\./, '');
    const row = t(table).get(id);
    const using = row ? updatePolicy(table, row) : false;
    if (using === null) return { status: 500, body: { code: 'NOT_MODELLED', message: `update ${table} not modelled` } };
    if (!row || !using) return { status: 200, body: [], rowsAffected: 0 };
    if (body.row_version != null && row.row_version != null && body.row_version < row.row_version) {
      return { status: 400, body: { code: 'MV001', message: 'stale_version', details: null, hint: null }, rowsAffected: 0 };
    }
    const next = { ...row, ...body };
    if (!updatePolicy(table, next)) return { ...rls(table, ' (USING expression)'), rowsAffected: 0 };
    t(table).set(id, next);
    return { status: 200, body: [{ id }], rowsAffected: 1 };
  };

  const rpc = (fn, args) => {
    if (fn === 'set_letter_mastery' || fn === 'set_skill_mastery') {
      const existing = receipts.get(args.p_correction_id);
      if (!existing) {
        if (!deliversToChild(args.p_child_id)) {
          return { status: 403, body: { code: '42501', details: null, hint: null, message: 'Active child delivery assignment required' } };
        }
        receipts.set(args.p_correction_id, { ...args, fn });
        const key = fn === 'set_letter_mastery'
          ? `${args.p_child_id}:${args.p_letter}:${args.p_language}` : `${args.p_child_id}:${args.p_step_key}`;
        t(fn === 'set_letter_mastery' ? 'letter_mastery' : 'skill_mastery').set(key, {
          child_id: args.p_child_id, confirmed: args.p_confirmed, changed_at: args.p_changed_at,
          correction_id: args.p_correction_id, changed_by: actor,
        });
      }
      return { status: 200, body: [{ outcome: 'applied', confirmed: args.p_confirmed, changed_at: args.p_changed_at,
        correction_id: args.p_correction_id, changed_by: actor }] };
    }
    if (fn === 'report_mobile_support_incident_v2') {
      t('mobile_support_incidents').set(String(t('mobile_support_incidents').size + 1), args.p_envelope);
      return { status: 200, body: { status: 'inserted' } };
    }
    return { status: 404, body: { code: 'PGRST202', message: `rpc ${fn} not modelled` } };
  };

  const fetchImpl = async (input, init = {}) => {
    const url = new URL(typeof input === 'string' ? input : input.url);
    const method = String(init.method || 'GET').toUpperCase();
    const headers = new Headers(init.headers);
    const prefer = headers.get('prefer') || '';
    const body = init.body ? JSON.parse(init.body) : undefined;
    const target = url.pathname.replace(/^\/rest\/v1\//, '');
    const entry = { seq: log.length + 1, method, target, prefer, onConflict: url.searchParams.get('on_conflict'),
      query: url.search, rawBody: body };
    let result;
    if (target.startsWith('rpc/')) {
      const fn = target.slice(4);
      entry.op = 'rpc';
      entry.fn = fn;
      if (fn === 'set_letter_mastery') entry.childId = body.p_child_id;
      result = rpc(fn, body);
    } else if (method === 'POST') {
      entry.op = Array.isArray(body) ? `upsert[batch:${body.length}]` : 'upsert';
      entry.table = target;
      entry.ids = (Array.isArray(body) ? body : [body]).map(r => r.id);
      entry.childIds = (Array.isArray(body) ? body : [body]).map(r => r.child_id).filter(Boolean);
      result = upsert(target, body, prefer);
    } else if (method === 'PATCH') {
      entry.op = 'update';
      entry.table = target;
      entry.ids = [url.searchParams.get('id')];
      entry.patch = body;
      entry.select = url.searchParams.get('select');
      result = patch(target, body, url.searchParams);
    } else {
      entry.op = method.toLowerCase();
      entry.table = target;
      result = { status: 200, body: [] };
    }
    entry.status = result.status;
    entry.outcome = result.status < 300 ? 'ok' : result.body?.code;
    if (result.status >= 300) entry.message = result.body?.message;
    if (result.rowsAffected != null) entry.rowsAffected = result.rowsAffected;
    log.push(entry);
    return new Response(result.body === undefined ? null : JSON.stringify(result.body), {
      status: result.status, headers: { 'content-type': 'application/json' },
    });
  };

  const seed = (table, row) => t(table).set(row.id, { ...row });
  const snapshot = () => JSON.parse(JSON.stringify(Object.fromEntries(Object.entries(rows).map(([k, m]) => [k, [...m.values()]]))));
  return { fetchImpl, rows, t, log, seed, snapshot, flags, writeForChild, deliversToChild };
};
